8def filter_vacancies(vacancies: List[Vacancy], words: List[str]) -> List[Vacancy]: 

9 """ 

10 Фильтрует вакансии по ключевым словам в описании или названии. 

11 """ 

12 lowered = [w.lower().strip() for w in words if w.strip()] 

13 if not lowered: 

14 return vacancies 

15 

16 result: List[Vacancy] = []

24def parse_salary_range(s: str) -> Tuple[int, int]: 

25 """ 

26 Парсит диапазон зарплат из строки вида '100000-150000' или '100000 150000'. 

27 Возвращает (min_salary, max_salary). 

28 """ 

29 cleaned = s.replace("–", "-").replace("—", "-").replace(" ", "") 

30 if "-" not in cleaned: 

31 value = int(cleaned) if cleaned.isdigit() else 0 

32 return value, 10**18 

33 

34 left, right = cleaned.split("-", 1) 

35 min_salary = int(left) if left.isdigit() else 0 

36 max_salary = int(right) if right.isdigit() else 10**18 

37 return min_salary, max_salary
